💻 Ficha Técnica - ComboBox Lombok

ExamFormController.java 
// Muestra el nombre, no la dirección de memoria
Callback<ListView<Subject>, ListCell<Subject>> cellFactory = lv -> new ListCell<>() {
    @Override
    protected void updateItem(Subject item, boolean empty) {
        super.updateItem(item, empty);
        setText(empty || item == null ? "" : item.getNameSubject());
    }
};
subjectCombo.setCellFactory(cellFactory);
subjectCombo.setButtonCell(cellFactory.call(null));
